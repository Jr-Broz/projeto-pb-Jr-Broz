=== ros2 interface show percepcao_aracnideos_interfaces/msg/DeteccaoAracnideo ===
# Uma deteccao de aracnideo num frame.
# Campos de caixa em pixels, canto superior esquerdo (x, y) + largura e altura.

string  classe            # rotulo predito: viuva-negra, teia-dourada, ...
float32 confianca         # 0.0 a 1.0
int32   x                 # canto superior esquerdo da bbox
int32   y
int32   w                 # largura da bbox
int32   h                 # altura da bbox

int32   id_rastreio       # id estavel atribuido pelo rastreador (-1 se nao rastreado)
int32   cx_suavizado      # centroide apos o filtro de Kalman
int32   cy_suavizado
float32 velocidade_px_s   # modulo da velocidade estimada pelo filtro

std_msgs/Header               header
	builtin_interfaces/Time stamp
		int32 sec
		uint32 nanosec
	string frame_id
DeteccaoAracnideo[]           deteccoes
	string  classe            #
	float32 confianca         #
	int32   x                 #
	int32   y
	int32   w                 #
	int32   h                 #
	int32   id_rastreio       #
	int32   cx_suavizado      #
	int32   cy_suavizado
	float32 velocidade_px_s   #
int32                         total
float32                       limiar_confianca_usado
string                        modelo                # ex.: yolov8n.onnx ou hsv-fallback
float32                       tempo_inferencia_ms

# ---- instrumentacao de robustez ----
# Um sistema que SABE que nao esta enxergando bem vale mais que um que
# devolve silencio. Estes campos deixam a condicao da imagem auditavel
# por qualquer no a jusante, sem precisar olhar a imagem.
float32                       qualidade_score       # 0.0 (ruim) a 1.0 (limpa)
float32                       brilho_medio          # 0-255, canal L
float32                       contraste             # desvio padrao de L
float32                       nitidez               # variancia do Laplaciano
string                        problemas_imagem      # "subexposta,desfocada" ou ""
string                        preproc_aplicado      # etapas aplicadas, ou ""
int32                         resgatadas_histerese  # deteccoes fracas aceitas
int32                         rejeitadas_histerese  # deteccoes fracas descartadas

=== ros2 interface show percepcao_aracnideos_interfaces/action/VarrerCena ===
# Acao principal do projeto: varrer a cena por um tempo determinado,
# acumulando deteccoes e devolvendo metricas ao final.
# Suporta cancelamento a qualquer momento.

# ---------- GOAL ----------
float32 duracao_s              # por quanto tempo varrer
string  classe_alvo            # "" = qualquer classe
float32 limiar_confianca       # limiar so para esta varredura; -1.0 = usar o do no
---
# ---------- RESULT ----------
bool    sucesso
string  mensagem
int32   frames_processados
int32   deteccoes_totais
int32   alvos_unicos           # ids de rastreio distintos vistos
float32 confianca_media
float32 confianca_maxima
float32 duracao_real_s
float32 fps_medio
string[] classes_vistas
int32[]  contagem_por_classe
---
# ---------- FEEDBACK ----------
float32 progresso              # 0.0 a 1.0
float32 tempo_decorrido_s
int32   frames_processados
int32   deteccoes_no_frame
int32   deteccoes_acumuladas
float32 melhor_confianca
string  estado                 # "varrendo", "cancelando", ...
